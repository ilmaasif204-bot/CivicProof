// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract ComplaintLedger {

    enum Status {
        Created,
        Acknowledged,
        ProofSubmitted,
        Resolved,
        Disputed,
        Reopened
    }

    struct Event {
        Status status;
        address actor;
        string  evidenceHash;
        string  metaHash;
        uint256 blockTimestamp;
    }

    struct Complaint {
        address citizen;
        string  category;
        string  locationHash;
        Event[] history;
        uint256 confirmVotes;
        bool    exists;
    }

    mapping(uint256 => Complaint) private complaints;
    uint256 public nextComplaintId;

    mapping(uint256 => mapping(address => bool)) public hasVoted;

    uint256 public constant CONFIRM_THRESHOLD = 3;

    event ComplaintFiled(uint256 indexed id, address indexed citizen, string category);
    event StatusChanged(uint256 indexed id, Status newStatus, address indexed actor, uint256 timestamp);

    function fileComplaint(
        string calldata category,
        string calldata locationHash,
        string calldata evidenceHash
    ) external returns (uint256) {
        uint256 id = nextComplaintId++;
        Complaint storage c = complaints[id];
        c.citizen = msg.sender;
        c.category = category;
        c.locationHash = locationHash;
        c.exists = true;

        c.history.push(Event({
            status: Status.Created,
            actor: msg.sender,
            evidenceHash: evidenceHash,
            metaHash: "",
            blockTimestamp: block.timestamp
        }));

        emit ComplaintFiled(id, msg.sender, category);
        emit StatusChanged(id, Status.Created, msg.sender, block.timestamp);
        return id;
    }

    function acknowledge(uint256 id) external {
        Complaint storage c = _getExisting(id);
        Status current = _currentStatus(c);
        require(
            current == Status.Created || current == Status.Reopened,
            "Illegal transition: can only acknowledge a new or reopened complaint"
        );
        _append(c, id, Status.Acknowledged, "", "");
    }

    function submitProof(
        uint256 id,
        string calldata afterEvidenceHash,
        string calldata metaHash
    ) external {
        Complaint storage c = _getExisting(id);
        Status current = _currentStatus(c);
        require(
            current == Status.Acknowledged,
            "Illegal transition: must be acknowledged before proof can be submitted"
        );
        require(bytes(afterEvidenceHash).length > 0, "Evidence hash required - no empty proof");

        _append(c, id, Status.ProofSubmitted, afterEvidenceHash, metaHash);

        c.confirmVotes = 0;
    }

    function confirmResolution(uint256 id) external {
        Complaint storage c = _getExisting(id);
        Status current = _currentStatus(c);
        require(current == Status.ProofSubmitted, "No pending proof to confirm");
        require(!hasVoted[id][msg.sender], "Already voted on this proof round");

        hasVoted[id][msg.sender] = true;
        c.confirmVotes++;

        if (c.confirmVotes >= CONFIRM_THRESHOLD) {
            _append(c, id, Status.Resolved, "", "");
        }
    }

    function disputeResolution(uint256 id, string calldata reasonHash) external {
        Complaint storage c = _getExisting(id);
        Status current = _currentStatus(c);
        require(
            current == Status.ProofSubmitted || current == Status.Resolved,
            "Can only dispute a submitted or already-resolved proof"
        );
        _append(c, id, Status.Disputed, "", reasonHash);
    }

    function reopen(uint256 id) external {
        Complaint storage c = _getExisting(id);
        Status current = _currentStatus(c);
        require(current == Status.Disputed, "Can only reopen a disputed complaint");
        _append(c, id, Status.Reopened, "", "");
    }

    function getHistory(uint256 id) external view returns (Event[] memory) {
        return _getExisting(id).history;
    }

    function getCurrentStatus(uint256 id) external view returns (Status) {
        return _currentStatus(_getExisting(id));
    }

    function getComplaintCount() external view returns (uint256) {
        return nextComplaintId;
    }

    function _currentStatus(Complaint storage c) internal view returns (Status) {
        return c.history[c.history.length - 1].status;
    }

    function _getExisting(uint256 id) internal view returns (Complaint storage) {
        Complaint storage c = complaints[id];
        require(c.exists, "Complaint does not exist");
        return c;
    }

    function _append(
        Complaint storage c,
        uint256 id,
        Status newStatus,
        string memory evidenceHash,
        string memory metaHash
    ) internal {
        c.history.push(Event({
            status: newStatus,
            actor: msg.sender,
            evidenceHash: evidenceHash,
            metaHash: metaHash,
            blockTimestamp: block.timestamp
        }));
        emit StatusChanged(id, newStatus, msg.sender, block.timestamp);
    }
}
